    foreach(asset IN ITEMS environmentSpecular.env brdf-lut.png)
        if(NOT EXISTS "${LITE_SANDBOX_ASSET_DIRECTORY}/${asset}")
            message(FATAL_ERROR "LiteSandbox lighting asset missing: ${asset}")
        endif()
        add_custom_command(TARGET LiteSandbox POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:LiteSandbox>/Assets"
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${LITE_SANDBOX_ASSET_DIRECTORY}/${asset}" "$<TARGET_FILE_DIR:LiteSandbox>/Assets/${asset}")
    endforeach()
    if(EXISTS "${LITE_SANDBOX_ASSET_DIRECTORY}/Notices")
        add_custom_command(TARGET LiteSandbox POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_directory
                "${LITE_SANDBOX_ASSET_DIRECTORY}/Notices" "$<TARGET_FILE_DIR:LiteSandbox>/Assets/Notices")
    endif()
endif()
