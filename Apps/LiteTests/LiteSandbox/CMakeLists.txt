if(NOT WIN32)
    message(FATAL_ERROR "The initial visible LiteSandbox host requires Win32.")
endif()
if(NOT TARGET LiteLayer OR NOT TARGET LiteShaderCompiler OR NOT TARGET LiteAssetDecoder)
    message(FATAL_ERROR "LiteSandbox requires LiteLayer, LiteShaderCompiler and LiteAssetDecoder.")
endif()
add_executable(LiteSandbox Sandbox.cpp AssetIO.cpp)
set(LITE_SANDBOX_OUTPUT_DIRECTORY "" CACHE PATH "Optional sandbox binary directory.")
if(LITE_SANDBOX_OUTPUT_DIRECTORY)
    set_target_properties(LiteSandbox PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${LITE_SANDBOX_OUTPUT_DIRECTORY}")
endif()
target_compile_features(LiteSandbox PRIVATE cxx_std_20)
target_compile_definitions(LiteSandbox PRIVATE UNICODE _UNICODE)
target_link_libraries(LiteSandbox PRIVATE LiteLayer LiteShaderCompiler LiteAssetDecoder bgfx
    user32 shell32 windowscodecs ole32)
target_link_options(LiteSandbox PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/LiteSandbox.map")
warnings_as_errors(LiteSandbox)
add_executable(LiteAssetTextureTests ../AssetTests/GpuTextures.cpp AssetIO.cpp)
target_compile_features(LiteAssetTextureTests PRIVATE cxx_std_20)
target_include_directories(LiteAssetTextureTests PRIVATE "${PROJECT_SOURCE_DIR}/Core/LiteLayer/Source")
target_link_libraries(LiteAssetTextureTests PRIVATE LiteLayer LiteAssetDecoder bgfx
    user32 windowscodecs ole32 d3d11)
warnings_as_errors(LiteAssetTextureTests)
add_executable(LiteAssetDecoder3Tests ../AssetTests/DecoderV3.cpp)
target_link_libraries(LiteAssetDecoder3Tests PRIVATE LiteAssetDecoder)
target_include_directories(LiteAssetDecoder3Tests PRIVATE "${PROJECT_SOURCE_DIR}/Core/LiteLayer/Include")
target_compile_features(LiteAssetDecoder3Tests PRIVATE cxx_std_17)
warnings_as_errors(LiteAssetDecoder3Tests)
add_test(NAME LiteAssetDecoder3Tests COMMAND LiteAssetDecoder3Tests)
set_property(TARGET LiteSandbox PROPERTY FOLDER Apps/LiteTests)
set(LITE_SANDBOX_ASSET_DIRECTORY "" CACHE PATH "Existing licensed environment and BRDF source directory.")
if(LITE_SANDBOX_ASSET_DIRECTORY)
    foreach(asset IN ITEMS environmentSpecular.env brdf-lut.png)
        if(NOT EXISTS "${LITE_SANDBOX_ASSET_DIRECTORY}/${asset}")
            message(FATAL_ERROR "LiteSandbox lighting asset missing: ${asset}")
        endif()
        add_custom_command(TARGET LiteSandbox POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:LiteSandbox>/Assets"
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${LITE_SANDBOX_ASSET_DIRECTORY}/${asset}" "$<TARGET_FILE_DIR:LiteSandbox>/Assets/${asset}")
    endforeach()
endif()
